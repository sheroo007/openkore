# SimpleTeleHunter v4.5.1 - Configuration Example
# ================================================
# 
# A smart teleport hunting plugin with instant teleport support
#
# FEATURES:
# - Automatic teleport when no relevant monsters nearby
# - Instant re-teleport if landing on empty/occupied spots
# - Smart monster counting (only counts monsters you're fighting or clean monsters)
# - Grace period after teleport to allow monsters to load
#
# INSTALLATION:
# 1. Place SimpleTeleHunter.pl in your plugins/ folder
# 2. Add the configuration below to your config.txt
# 3. Reload plugins with "reload plugins" command
#
# ================================================

# Enable/disable the plugin
simpleTeleHunter_enabled 1

# Time to wait (in seconds) before teleporting when idle
simpleTeleHunter_idleTime 5

# Radius to check for monsters for instant teleport (after landing)
simpleTeleHunter_instantRadius 7

# Grace period (in seconds) to wait after teleport before checking
# This allows time for monsters to load properly
simpleTeleHunter_instantGracePeriod 0.5

# Radius to check for monsters during normal idle checks
simpleTeleHunter_checkRadius 10

# Teleport spell/skill name to use
simpleTeleHunter_teleportSpell Teleport

# Test mode (set to 1 to disable actual teleporting, useful for testing)
simpleTeleHunter_testMode 0
